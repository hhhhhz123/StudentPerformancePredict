comparison <- compare_models(stepwise_res, lasso_res, ridge_res)

##

5. Save Models

saveRDS(stepwise_res$model, "../models/stepwise_model.rds")
saveRDS(lasso_res$model, "../models/lasso_model.rds")
saveRDS(ridge_res$model, "../models/ridge_model.rds")

# Save best
best_name <- comparison$Model[1]
if (best_name == "Lasso") {
  saveRDS(lasso_res$model, "../models/final_model.rds")
} else if (best_name == "Ridge") {
  saveRDS(ridge_res$model, "../models/final_model.rds")
} else {
  saveRDS(stepwise_res$model, "../models/final_model.rds")
}

cat("Saved best model:", best_name, "\n")
## Saved best model: Stepwise
